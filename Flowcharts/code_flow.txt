  [MAX_PROBE_QUEUE_SIZE]   = 4
  [_PROBE_TYPE_PRIORITY]   = {how:0.90, decision:0.80, error:0.70,
                               edge_case:0.65, rationale:0.55, sequencing:0.50}
  [_FOLLOWUP_ALLOWED_TYPES] = {"how", "decision"}
  [_SINGLE_INSTANCE_TYPES]  = {"error", "edge_case"}

  class ScenarioRunner:

    __init__(scenario, model, api_key, base_url)
      self.phase           = "recall"
      self.history         = []
      self.user_inputs     = []
      self.writing_metrics = []
      self.recall_history  = []
      self.probe_history   = []
      self.probe_queue     = []
      self.probe_index     = 0
      self.probe_results   = {}
      self._probe_followup_pending = False

    @property recall_transcript
      --> "Examiner: <situation>\n\n<role>: <turn>\n\nExaminer: <ack>..."
      [built from recall_history]

    @property probe_transcript
      --> "\n\n".join(role + ": " + msg for msg in probe_history)

    start()
      --> reset all state (history, transcripts, probe state)
      --> return "Examiner: <situation>\n\n<framing prompt>"

    respond(user_input, writing_metrics=None)
      --> self.user_inputs.append(user_input)
      --> self.writing_metrics.append(writing_metrics or {})
      if phase == "recall"    --> _recall_respond(user_input)
      if phase == "probing"   --> _probe_respond(user_input)
      if phase == "concluded" --> return "", True

    end_recall()
      --> self.phase = "probing"
      --> self.probe_queue = _build_probe_queue()
      if probe_queue empty:
        --> self.phase = "concluded"; return "", True
      --> first_probe = _format_probe_delivery(probe_queue[0])
      --> probe_history.append(first_probe)
      --> return first_probe, False

    transcript()
      --> combines recall_history + probe_history with section label
      --> return full text

    probe_count()   --> len(probe_queue)
    current_probe_number() --> min(probe_index + 1, len(probe_queue))

    ── Recall phase ───────────────────────────────────────────────────────

    _recall_respond(user_input)
      --> recall_history.append({"role":"user", content})
      --> history.append(...)
      --> ack = _get_recall_ack(user_input)
      --> recall_history.append({"role":"assistant", ack})
      --> history.append(...)
      --> return ack, False

    _get_recall_ack(user_input)
      --> build minimal non-leading prompt
      --> llm.llm_generate(model, prompt, api_key, base_url)  *
      --> strip <think>...</think> tags
      --> take first sentence (up to first .!?)
      fallback on any error: FALLBACK_RECALL_ACK[n % 4]

    ── Gap analysis & probe queue ─────────────────────────────────────────

    _build_probe_queue()
      --> probe_bank = scenario.get("probe_bank", [])
      --> normalise probe dicts: {probe_type, probe_text, target_key_point, success_criteria}
      if probe_bank empty:
        --> _generate_dynamic_probes(recall_txt, expert_answer, key_points)
      --> coverage_map, recall_disordered = _analyse_coverage(recall_txt, probe_bank, key_points)
      --> how_targets = {p["target_key_point"] for p of type "how"}
      for each probe in probe_bank:
        --> skip covered how/sequencing probes
        --> skip sequencing if not recall_disordered
        --> skip rationale if same target already has a how probe
        --> base_score = _PROBE_TYPE_PRIORITY[type]
        --> score += 0.08 if missing, +0.04 if partial
        --> add to candidates list
      --> sort candidates by score descending
      --> select top MAX_PROBE_QUEUE_SIZE, respecting _SINGLE_INSTANCE_TYPES caps
      --> annotate each: status="pending", exchange=[], _followup_sent=False
      --> return queue

    _analyse_coverage(recall_txt, probe_bank, key_points)
      [cached LLM call]
      --> build prompt with recall transcript + numbered probe summaries
      --> system: "assess which probes are already covered in recall"
      --> cached_evaluative_call(..., _PROMPT_VERSION_COVERAGE_ANALYSIS,
                                 system, prompt, _call)  *~
              _call: llm_chat_json(model, system, prompt, api_key, base_url,
